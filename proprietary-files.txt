# Keep MiuiCamera on /product: its privileged permission allowlist lives on the
# same partition, and Android 17 resolves those allowlists per partition. An
# app on /system with an allowlist on /product fails systemReady with
# "Signature|privileged permissions not in privileged permission allowlist".
product/priv-app/MiuiCamera/MiuiCamera.apk
